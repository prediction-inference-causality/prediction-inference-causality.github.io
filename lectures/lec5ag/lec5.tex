\frame{
\frametitle{Review}

\begin{itemize}
\item In the previous lecture, we discussed learning about subpopulation proportions and differences in subpopulation proportions based on a random sample (with replacement) from the population. \pause
\item We saw that our sample subpopulation proportion was likely to be somewhat close to the subpopulation proportion (lower sample size). \pause
\item But we also demonstrated that we could accurately predict how close we might by using confidence intervals. \pause
\item We also saw that our sample difference in subpopulation proportions was likely to be somewhat close to the population difference in subpopulation proportions. \pause
\item Again, we also demonstrated that we could predict how close we might by using confidence intervals. \pause
\item We used a bootstrapping approach to confidence intervals and showed how bootstrapping differences allowed us to generate numbers that were close to the numbers in a regression table.
\end{itemize}

}

\frame{
\frametitle{In this lecture}

\begin{itemize}
\item In this lecture, we discuss learning subpopulation means and differences between subpopulation means based on a random sample (with replacement) from the population. \pause
\item Returning to the CPS example, we will consider college degree incomes and the difference between college degree and non-college degree incomes. \pause
\item Even when the overall sample size is fixed (2258 for our example), different random samples will produce different subgroup sample sizes, and this adds a layer of complexity.  \pause
\item This complexity will not complicate our analysis much but it does complicate the justification for the analysis.
\end{itemize}

}

\frame{


Property \#1: {\it The sampling distribution of the subsample mean is approximately normal when $\sum_{i=1}^{2258}X_{i}$ is large.}
\begin{itemize}
\item The approximate normality result comes from a central limit theorem (CLT). 
\item We know that $\frac{\sum_{i=1}^{2258}Y_{i}X_{i}}{\sum_{i=1}^{2258}X_{i}}  \sim^{approx} N(?,?)$, but still don't know the mean or the variance of this distribution. 
\end{itemize}

}

\frame{
Property \#2: {\it The subsample mean is unbiased for the subpopulation mean.} %The expected value of the sample mean,
%$\E[\frac{\sum_{i=1}^{2258}Y_{i}X_{i}}{\sum_{i=1}^{2258}X_{i}}] = \mu_1$.

\vspace{0.2in}

Part 1 of Proof (condition on $X_1=x_1, X_2=x_2,..., X_{2258}=x_{2258}$):
\begin{align*}
\E[&\frac{\sum_{i=1}^{2258}Y_{i}X_{i}}{\sum_{i=1}^{2258}X_{i}}| X_1=x_1, X_2=x_2,..., X_{2258}=x_{2258}] \\
&=\frac{1}{\sum_{i=1}^{2258}x_{i}}\E[\sum_{i=1}^{2258}Y_{i}X_{i}| X_1=x_1, X_2=x_2,..., X_{2258}=x_{2258}]  \\
&=\frac{1}{\sum_{i=1}^{2258}x_{i}} \sum_{i=1}^{2258} \E[Y_{i}X_{i}| X_1=x_1, X_2=x_2,..., X_{2258}=x_{2258}]  \\
&=\frac{1}{\sum_{i=1}^{2258}x_{i}} \sum_{i=1}^{2258} x_i\E[Y_{i}| X_1=x_1, X_2=x_2,..., X_{2258}=x_{2258}] 
\end{align*}

}


\frame{
Property \#2: {\it The subsample mean is unbiased for the subpopulation mean.} %The expected value of the sample mean,
%$\E[\frac{\sum_{i=1}^{2258}Y_{i}X_{i}}{\sum_{i=1}^{2258}X_{i}}] = \mu_1$.

\vspace{0.2in}

Part 2 of Proof (condition on $X_1=x_1, X_2=x_2,..., X_{2258}=x_{2258}$):
\begin{align*}
\frac{1}{\sum_{i=1}^{2258}x_{i}}& \sum_{i=1}^{2258} x_i\E[Y_{i}| X_1=x_1, X_2=x_2,..., X_{2258}=x_{2258}]  \\
&= \frac{1}{\sum_{i=1}^{2258}x_{i}} \sum_{i=1}^{2258} x_i\E[Y_{i}| X_i=x_i]  \\
&= \frac{1}{\sum_{i=1}^{2258}x_{i}} \sum_{i=1}^{2258} x_i[x_i\mu_1 + (1-x_i)\mu_0] \\
&=\frac{1}{\sum_{i=1}^{2258}x_{i}} \sum_{i=1}^{2258} x_i \mu_1\\
& =\frac{1}{\sum_{i=1}^{2258}x_{i}} \mu_1 \sum_{i=1}^{2258} x_i = \mu_1.
\end{align*}

}




\frame{
Property \#2: {\it The subsample mean is unbiased for the subpopulation mean.} %The expected value of the sample mean,
%$\E[\frac{\sum_{i=1}^{2258}Y_{i}X_{i}}{\sum_{i=1}^{2258}X_{i}}] = \mu_1$.

\vspace{0.2in}

Part 3 of Proof:
\begin{align*}
\E[&\frac{\sum_{i=1}^{2258}Y_{i}X_{i}}{\sum_{i=1}^{2258}X_{i}}] \\
&=\E_{\bold{X}}[\E[\frac{\sum_{i=1}^{2258}Y_{i}X_{i}}{\sum_{i=1}^{2258}X_{i}}| X_1=x_1, X_2=x_2,..., X_{2258}=x_{2258}] ]\\
&=\E_{\bold{X}}[\mu_1]\\
&=\mu_1
\end{align*}

}

\frame{
Property \#3: The {\it sampling variance} of the subsample mean,

\vspace{0.2in}

Part 1 of Proof:
\begin{align*}
V[&\frac{\sum_{i=1}^{2258}Y_{i}X_{i}}{\sum_{i=1}^{2258}X_{i}}] \\
&= \E_{\bold{X}}[V[ \frac{\sum_{i=1}^{2258}Y_{i}X_{i}}{\sum_{i=1}^{2258}X_{i}}| X_1=x_1, X_2=x_2,..., X_{2258}=x_{2258} ] ]\\
&+ V_{\bold{X}}[\E[\frac{\sum_{i=1}^{2258}Y_{i}X_{i}}{\sum_{i=1}^{2258}X_{i}}| X_1=x_1, X_2=x_2,..., X_{2258}=x_{2258} ] ] \\
&= \E_{\bold{X}}[V[ \frac{\sum_{i=1}^{2258}Y_{i}X_{i}}{\sum_{i=1}^{2258}X_{i}}| X_1=x_1, X_2=x_2,..., X_{2258}=x_{2258} ] ]\\
&+ V_{\bold{X}}[\mu_1 ]\\
\end{align*}

}

\frame{
Property \#3: The {\it sampling variance} of the subsample mean,

\vspace{0.2in}

Part 2 of Proof:
\begin{align*}
\E_{\bold{X}}[&V[ \frac{\sum_{i=1}^{2258}Y_{i}X_{i}}{\sum_{i=1}^{2258}X_{i}}| X_1=x_1, X_2=x_2,..., X_{2258}=x_{2258} ] ]\\
&= \E_{\bold{X}}[(\frac{1}{\sum_{i=1}^{2258}x_{i}})^2 V[\sum_{i=1}^{2258}Y_{i}X_{i}| X_1=x_1, X_2=x_2,..., X_{2258}=x_{2258}] ]\\
&=\E_{\bold{X}}[(\frac{1}{\sum_{i=1}^{2258}x_{i}})^2 \sum_{i=1}^{2258} V[Y_{i}X_{i}| X_1=x_1, X_2=x_2,..., X_{2258}=x_{2258}]]  \\
&=\E_{\bold{X}}[(\frac{1}{\sum_{i=1}^{2258}x_{i}})^2 \sum_{i=1}^{2258} x_i^2 V[Y_{i}| X_1=x_1, X_2=x_2,..., X_{2258}=x_{2258}]] 
\end{align*}

}


\frame{
Property \#3: The {\it sampling variance} of the subsample mean,

\vspace{0.2in}

Part 3 of Proof:
\begin{align*}
\E_{\bold{X}}[&(\frac{1}{\sum_{i=1}^{2258}x_{i}})^2 \sum_{i=1}^{2258} x_i^2 V[Y_{i}| X_1=x_1, X_2=x_2,..., X_{2258}=x_{2258}]] \\
&= \E_{\bold{X}}[(\frac{1}{\sum_{i=1}^{2258}x_{i}})^2 \sum_{i=1}^{2258} x_i^2 V[Y_{i}| X_i=x_i]] \\
&= \E_{\bold{X}}[(\frac{1}{\sum_{i=1}^{2258}x_{i}})^2 \sum_{i=1}^{2258} x_i^2 (x_i \sigma^2_1 + (1-x_i)\sigma^2_0)] \\
&= \E_{\bold{X}}[(\frac{1}{\sum_{i=1}^{2258}x_{i}})^2 \sum_{i=1}^{2258} x_i^2 \sigma^2_1] \\
&= \sigma^2_1\E_{\bold{X}}[\frac{1}{\sum_{i=1}^{2258}x_{i}}]
\end{align*}

}
